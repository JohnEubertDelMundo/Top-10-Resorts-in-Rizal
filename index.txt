    <script>
        var currentResort = 0;
        var resorts = document.querySelectorAll('.resort');
        var numberButtons = document.querySelectorAll('.number-buttons button');

        function showResort(number) {
            currentResort = number;

            for (var i = 0; i < resorts.length; i++) {
                resorts[i].hidden = (i !== currentResort);
                numberButtons[i].classList.toggle('active', i === currentResort);
            }
        }

        showResort(0);
    </script>
